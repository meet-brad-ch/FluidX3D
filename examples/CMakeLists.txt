# Examples kept outside this repository: each folder holds a CMakeLists.txt with add_fluidx3d_example() calls, built
# with this tree's core, Setup API and configurations (-DFLUIDX3D_EXTERNAL_EXAMPLES="C:/path/a;C:/path/b")
set(FLUIDX3D_EXTERNAL_EXAMPLES "" CACHE STRING "Folders outside the repository with add_fluidx3d_example() projects (a list)")
foreach(folder IN LISTS FLUIDX3D_EXTERNAL_EXAMPLES)
    if(NOT EXISTS ${folder}/CMakeLists.txt)
        message(FATAL_ERROR "FLUIDX3D_EXTERNAL_EXAMPLES: no CMakeLists.txt in ${folder}")
    endif()
    get_filename_component(name ${folder} NAME)
    add_subdirectory(${folder} ${CMAKE_BINARY_DIR}/external_examples/${name})
endforeach()
